# Shiroko's own CMake build, fed by the host build that `make tab5-build` makes first.
idf_component_register(LDFRAGMENTS linker.lf)
set(_root ${CMAKE_CURRENT_LIST_DIR}/../../../..)
set(SHIROKO_TAB5_HOST ${_root}/build/tab5-host CACHE PATH "Host build with generated/, fonts/ and vcpkg_installed/")
file(GLOB _utlist ${SHIROKO_TAB5_HOST}/vcpkg_installed/*/include/utlist.h)
if(NOT _utlist OR NOT EXISTS ${SHIROKO_TAB5_HOST}/generated/shr_gen_builtin_package.c)
  message(FATAL_ERROR "No host build in ${SHIROKO_TAB5_HOST}: run `make tab5-build`")
endif()
cmake_path(GET _utlist PARENT_PATH _inc)
set(UTHASH_INCLUDE_DIR ${_inc} CACHE PATH "uthash and xxhash headers (header-only)")
set(SHIROKO_GENERATED_DIR ${SHIROKO_TAB5_HOST}/generated CACHE PATH "")
# zstd's decoder, its sources installed by vcpkg (zstd-source), so stored and zstd packages both load; OFF: stored only.
option(SHIROKO_TAB5_ZSTD "Build zstd's decoder in (make tab5-build TAB5_ZSTD=ON|OFF)" ON)
if(SHIROKO_TAB5_ZSTD)
  file(GLOB _zstd ${SHIROKO_TAB5_HOST}/vcpkg_installed/*/share/zstd-source/zstd.h)
  if(NOT _zstd)
    message(FATAL_ERROR "No zstd sources in ${SHIROKO_TAB5_HOST}: run `make tab5-build`")
  endif()
  cmake_path(GET _zstd PARENT_PATH _zstd)
  file(GLOB _zsrc ${_zstd}/common/*.c ${_zstd}/decompress/*.c)
  add_library(shiroko_zstd STATIC ${_zsrc})
  target_include_directories(shiroko_zstd SYSTEM INTERFACE ${_zstd})
  target_compile_definitions(shiroko_zstd PRIVATE ZSTD_LEGACY_SUPPORT=0 ZSTD_DISABLE_ASM)
  target_compile_options(shiroko_zstd PRIVATE -w)
  set(SHIROKO_ZSTD ON)
  set(SHIROKO_ZSTD_TARGET shiroko_zstd)
else()
  set(SHIROKO_ZSTD OFF)
endif()
# ESP-IDF's project() wrapper leaves PROJECT_IS_TOP_LEVEL true inside sub-projects.
set(SHIROKO_BUILD_TESTS OFF CACHE BOOL "")
set(SHIROKO_BUILD_EXAMPLES OFF CACHE BOOL "")
set(CMAKE_SKIP_INSTALL_RULES ON) # its export set cannot name the IDF libraries every target links
add_subdirectory(${_root} shiroko EXCLUDE_FROM_ALL)
target_link_libraries(${COMPONENT_LIB} INTERFACE shiroko)
# The file source needs a 64-bit off_t (newlib's is 32-bit); packages are mapped from flash instead.
get_target_property(_src shiroko_port_software SOURCES)
list(FILTER _src EXCLUDE REGEX "file_source\\.c$")
set_target_properties(shiroko_port_software PROPERTIES SOURCES "${_src}")
# IDF builds with -Werror and wraps libc headers with #include_next, which -Wpedantic reports.
get_property(_parts GLOBAL PROPERTY SHIROKO_PARTS)
foreach(t IN LISTS _parts)
  target_compile_options(${t} PRIVATE -Wno-pedantic)
endforeach()
