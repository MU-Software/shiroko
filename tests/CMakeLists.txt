# Tests may use every part's internal headers. LIB: the library linked instead of shiroko.
function(shiroko_test name)
  cmake_parse_arguments(ARG "" "LIB" "" ${ARGN})
  if(NOT ARG_LIB)
    set(ARG_LIB shiroko)
  endif()
  add_executable(${name} ${ARG_UNPARSED_ARGUMENTS})
  target_link_libraries(${name} PRIVATE ${ARG_LIB})
  target_include_directories(${name} SYSTEM PRIVATE ${GREATEST_INCLUDE_DIR} ${UTHASH_INCLUDE_DIR})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/unit ${PROJECT_SOURCE_DIR}/cores/shared
    ${PROJECT_SOURCE_DIR}/cores/compositor ${PROJECT_SOURCE_DIR}/cores/pl_res_bitmap_font ${PROJECT_SOURCE_DIR}/cores/pl_res_image
    ${PROJECT_SOURCE_DIR}/cores/pl_lyr_tilemap ${PROJECT_SOURCE_DIR}/ports/software)
  target_compile_definitions(${name} PRIVATE SHR_FONT_DIR="${CMAKE_BINARY_DIR}/fonts")
  target_compile_options(${name} PRIVATE ${SHIROKO_WARNINGS})
  add_dependencies(${name} shiroko_fonts)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
